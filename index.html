Chưa kết nối Webhook
Mẫu code Apps Script (Google Sheets > Tiện ích mở rộng > Apps Script): 
function doGet(e) {
  var props = PropertiesService.getScriptProperties();
  var raw = props.getProperty("QUAN_LY_TRO_STATE");
  return ContentService.createTextOutput(raw || "{}")
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var data = e.postData.contents;
  PropertiesService.getScriptProperties().setProperty("QUAN_LY_TRO_STATE", data);
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", time: new Date() }))
    .setMimeType(ContentService.MimeType.JSON);
}
